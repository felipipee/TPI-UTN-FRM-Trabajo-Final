Ordenamiento de burbuja y estadísticas implementados sin lambdas ni built-ins.

# funciones.py — Algoritmo de burbuja sin lambda

def _clave_orden(pais, campo):
    """Devuelve el valor de comparación según el campo."""
    if campo == "nombre":
        return pais["nombre"].lower()
    return pais[campo]


def ordenar_paises(paises, campo, ascendente=True):
    """Bubble sort sin sorted() ni lambda."""
    resultado = paises[:]            # copia, no modifica el original
    n = len(resultado)
    for i in range(n - 1):
        for j in range(n - i - 1):
            val_a = _clave_orden(resultado[j],     campo)
            val_b = _clave_orden(resultado[j + 1], campo)
            intercambiar = val_a > val_b if ascendente else val_a < val_b
            if intercambiar:
                resultado[j], resultado[j+1] = resultado[j+1], resultado[j]
    return resultado


# Estadísticas sin max() / min() con key=
def calcular_estadisticas(paises):
    total_pob, total_sup = 0, 0
    mayor = menor = paises[0]
    for p in paises:
        total_pob += p["poblacion"]
        total_sup += p["superficie"]
        if p["poblacion"] > mayor["poblacion"]: mayor = p
        if p["poblacion"] < menor["poblacion"]: menor = p
    return {
        "total":     len(paises),
        "mayor_pob": mayor,
        "menor_pob": menor,
        "prom_pob":  total_pob // len(paises),
        "prom_sup":  total_sup // len(paises),
    }
